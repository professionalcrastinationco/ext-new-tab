Add JavaScript to populate and handle icon picker:

1. Create function to populate icon picker:
```javascript
function populateIconPicker() {
  const picker = document.getElementById('icon-picker');
  const hiddenInput = document.getElementById('category-icon');

  // Get category icons from icons.js
  const iconNames = Object.keys(CATEGORY_ICONS);

  picker.innerHTML = iconNames.map(name => `
    <button type="button"
            class="icon-picker-item${hiddenInput.value === name ? ' selected' : ''}"
            data-icon="${name}"
            title="${name}">
      ${CATEGORY_ICONS[name]}
    </button>
  `).join('');

  // Click handler for icon selection
  picker.addEventListener('click', (e) => {
    const item = e.target.closest('.icon-picker-item');
    if (!item) return;

    // Update selection
    picker.querySelectorAll('.icon-picker-item').forEach(el => el.classList.remove('selected'));
    item.classList.add('selected');
    hiddenInput.value = item.dataset.icon;
  });
}
```

// Render category icon - handles both emoji (legacy) and Phosphor names
function renderCategoryIcon(iconValue) {
  // Check if it's a Phosphor icon name
  if (CATEGORY_ICONS && CATEGORY_ICONS[iconValue]) {
    return CATEGORY_ICONS[iconValue];
  }
  // Fallback: treat as emoji or text
  return iconValue || CATEGORY_ICONS['folder'];
}
